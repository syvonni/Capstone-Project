// Helper function to get event type label
export function getEventTypeLabel(eventType) {
  return EVENT_TYPE_LABELS[eventType] || eventType
    .split('_')
    .map(word => word.charAt(0).toUpperCase() + word.slice(1))
    .join(' ')
}
